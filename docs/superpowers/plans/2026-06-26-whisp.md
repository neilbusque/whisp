# Whisp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A local-first macOS voice dictation app — hold Fn, speak, get cleaned-up text pasted at the cursor in any app — with voice-triggered snippets (verbatim or AI-generated).

**Architecture:** Two SwiftPM targets. `WhispCore` holds all pure logic (config, snippets, transcript cleanup prompt-building, and the `SpeechPipeline` orchestration behind injected protocols) and is fully unit-tested. `Whisp` is the executable menu-bar app that supplies concrete implementations (AVAudioEngine recorder, vendored `whisper.cpp` transcriber, DeepSeek cleaner, paste-at-cursor inserter, Fn hotkey, floating HUD, SwiftUI managers).

**Tech Stack:** Swift 6, SwiftPM, Command Line Tools only (no Xcode.app), Swift Testing, AppKit/SwiftUI/AVFoundation/CoreGraphics/Carbon, vendored `whisper.cpp` (C/C++/Metal), DeepSeek (OpenAI-compatible SSE).

## Global Constraints

- Swift 6, **Command Line Tools only** — no Xcode.app, no `.xcodeproj`. Build via `swift build` and `build.sh`.
- **No fetched package dependencies.** `whisper.cpp` is *vendored source* in-repo, compiled as a C target.
- Two targets only: `WhispCore` (logic + tests) and `Whisp` (executable, `LSUIElement`).
- Bundle id: `com.busqueneil.whisp`. App name: **Whisp**. Repo: `~/whisp`. Local git, **no GitHub remote**.
- Config: `~/.whisp/config.json`. Snippets: `~/.whisp/snippets.json`. Models: `~/.whisp/models/`. All gitignored; `config.example.json` committed.
- Git author: `Neil Busque <busqueneil@gmail.com>`. Commit messages end with the Co-Authored-By trailer used in this repo.
- Audio stays on-device (Whisper local). Only transcript text goes to DeepSeek.
- English v1; default model `ggml-base.en`.
- Self-signed signing cert in a dedicated keychain so Mic + Accessibility grants persist across rebuilds.

```swift
import Foundation

public final class SnippetStore {
    private(set) public var snippets: [Snippet]

    public init(snippets: [Snippet]) { self.snippets = snippets }

    public var all: [Snippet] { snippets }

    public static func load(from data: Data) throws -> SnippetStore {
        SnippetStore(snippets: try JSONDecoder().decode([Snippet].self, from: data))
    }

    public func encoded() throws -> Data {
        let enc = JSONEncoder()
        enc.outputFormatting = [.prettyPrinted, .sortedKeys]
        return try enc.encode(snippets)
    }

    public func upsert(_ s: Snippet) {
        if let i = snippets.firstIndex(where: { $0.id == s.id }) { snippets[i] = s }
        else { snippets.append(s) }
    }

    public func delete(id: UUID) {
        snippets.removeAll { $0.id == id }
    }

    /// Lowercase, strip punctuation to spaces, collapse whitespace, trim.
    public static func normalize(_ s: String) -> String {
        let lowered = s.lowercased()
        let mapped = lowered.unicodeScalars.map { scalar -> Character in
            if CharacterSet.alphanumerics.contains(scalar) { return Character(scalar) }
            return " "
        }
        let collapsed = String(mapped).split(separator: " ", omittingEmptySubsequences: true)
        return collapsed.joined(separator: " ")
    }

    /// Start-anchored match: the normalized transcript must equal the keyword,
    /// or begin with `keyword + " "`. Returns the matched snippet + trailing remainder.
    public func match(transcript: String) -> SnippetMatch? {
        let t = SnippetStore.normalize(transcript)
        for s in snippets where s.enabled {
            let k = SnippetStore.normalize(s.keyword)
            guard !k.isEmpty else { continue }
            if t == k {
                return SnippetMatch(snippet: s, remainder: "")
            }
            if t.hasPrefix(k + " ") {
                let remainder = String(t.dropFirst(k.count + 1))
                return SnippetMatch(snippet: s, remainder: remainder)
            }
        }
        return nil
    }
}
```

```swift
import Foundation

public struct ChatMessage: Equatable, Codable, Sendable {
    public let role: String
    public let content: String
    public init(role: String, content: String) { self.role = role; self.content = content }
}

public enum TranscriptCleaner {
    static let fillers: Set<String> = ["um", "uh", "erm", "ah", "like", "you know", "i mean"]

    public static func polishPrompt(raw: String) -> [ChatMessage] {
        let system = """
        You clean up dictated speech into polished written text. Remove filler words \
        (um, uh, like), fix grammar and punctuation, and format naturally. Do NOT add \
        content, greetings, or commentary. Do NOT use em dashes. Return ONLY the cleaned text.
        """
        return [
            ChatMessage(role: "system", content: system),
            ChatMessage(role: "user", content: raw),
        ]
    }

    public static func aiTemplatePrompt(template: String, dictation: String) -> [ChatMessage] {
        let system = """
        You generate written text from a template instruction plus the user's spoken \
        details. Follow the instruction exactly. Do NOT use em dashes. Return ONLY the \
        generated text, no commentary.
        """
        let user = """
        INSTRUCTION:
        \(template)

        DETAILS:
        \(dictation)
        """
        return [
            ChatMessage(role: "system", content: system),
            ChatMessage(role: "user", content: user),
        ]
    }

    /// Pure on-device fallback used when DeepSeek is unreachable.
    public static func offlineClean(_ raw: String) -> String {
        var words = raw.split(separator: " ").map(String.init)
        words.removeAll { fillers.contains($0.lowercased()) }
        guard !words.isEmpty else { return "" }
        var text = words.joined(separator: " ")
        // Capitalize first letter.
        text = text.prefix(1).uppercased() + text.dropFirst()
        // Add terminal punctuation if missing.
        if let last = text.last, !".?!".contains(last) { text += "." }
        return text
    }
}
```

```swift
import Foundation

public enum PipelineStatus: Equatable, Sendable {
    case idle, listening, transcribing, polishing
    case done(String)
    case error(String)
}

public protocol Transcribing: Sendable {
    func transcribe(_ samples: [Float]) async throws -> String
}
public protocol Cleaning: Sendable {
    func complete(_ messages: [ChatMessage]) async throws -> String
}
public protocol TextInserting: Sendable {
    func insert(_ text: String) throws
}

public final class SpeechPipeline: @unchecked Sendable {
    private let snippets: SnippetStore
    private let transcriber: Transcribing
    private let cleaner: Cleaning
    private let inserter: TextInserting
    private let onStatus: (PipelineStatus) -> Void

    public init(snippets: SnippetStore, transcriber: Transcribing, cleaner: Cleaning,
                inserter: TextInserting, onStatus: @escaping (PipelineStatus) -> Void) {
        self.snippets = snippets
        self.transcriber = transcriber
        self.cleaner = cleaner
        self.inserter = inserter
        self.onStatus = onStatus
    }

    public func process(samples: [Float]) async {
        do {
            onStatus(.transcribing)
            let raw = try await transcriber.transcribe(samples)
            let trimmed = raw.trimmingCharacters(in: .whitespacesAndNewlines)
            guard !trimmed.isEmpty else {
                onStatus(.error("Nothing heard"))
                return
            }

            let output: String
            if let m = snippets.match(transcript: trimmed) {
                switch m.snippet.mode {
                case .verbatim:
                    output = m.snippet.content
                case .ai:
                    onStatus(.polishing)
                    output = try await cleanOrFallback(
                        TranscriptCleaner.aiTemplatePrompt(template: m.snippet.content,
                                                           dictation: m.remainder),
                        fallback: m.remainder.isEmpty ? m.snippet.content : m.remainder)
                }
            } else {
                onStatus(.polishing)
                output = try await cleanOrFallback(
                    TranscriptCleaner.polishPrompt(raw: trimmed),
                    fallback: TranscriptCleaner.offlineClean(trimmed))
            }

            try inserter.insert(output)
            onStatus(.done(output))
        } catch {
            onStatus(.error(error.localizedDescription))
        }
    }

    private func cleanOrFallback(_ messages: [ChatMessage], fallback: String) async throws -> String {
        do {
            let result = try await cleaner.complete(messages)
            let trimmed = result.trimmingCharacters(in: .whitespacesAndNewlines)
            return trimmed.isEmpty ? fallback : trimmed
        } catch {
            return fallback
        }
    }
}
```

```bash
# Creates a self-signed code-signing identity in a dedicated keychain so that
# Microphone + Accessibility TCC grants persist across rebuilds.
set -euo pipefail
KEYCHAIN="$HOME/Library/Keychains/whisp-signing.keychain-db"
PASS="whisp"
IDENTITY="Whisp Signing"

if security find-identity -v "$KEYCHAIN" 2>/dev/null | grep -q "$IDENTITY"; then
  echo "Signing identity already present."
  exit 0
fi

TMP="$(mktemp -d)"
# Use Homebrew openssl (-legacy) so the .p12 is importable by macOS security.
OPENSSL="$(command -v /opt/homebrew/opt/openssl@3/bin/openssl || command -v openssl)"
"$OPENSSL" req -x509 -newkey rsa:2048 -keyout "$TMP/k.key" -out "$TMP/c.crt" \
  -days 3650 -nodes -subj "/CN=$IDENTITY" \
  -addext "extendedKeyUsage=codeSigning" >/dev/null 2>&1
"$OPENSSL" pkcs12 -export -legacy -inkey "$TMP/k.key" -in "$TMP/c.crt" \
  -out "$TMP/c.p12" -passout pass:"$PASS" >/dev/null 2>&1

security create-keychain -p "$PASS" "$KEYCHAIN" 2>/dev/null || true
security unlock-keychain -p "$PASS" "$KEYCHAIN"
security import "$TMP/c.p12" -k "$KEYCHAIN" -P "$PASS" -T /usr/bin/codesign
security set-key-partition-list -S apple-tool:,apple:,codesign: -s -k "$PASS" "$KEYCHAIN" >/dev/null
security list-keychains -d user -s "$KEYCHAIN" $(security list-keychains -d user | tr -d '"')
rm -rf "$TMP"
echo "Created signing identity '$IDENTITY' in $KEYCHAIN"
```

```swift
import Foundation
import CWhisper
import WhispCore

final class WhisperTranscriber: Transcribing, @unchecked Sendable {
    private let ctx: OpaquePointer
    private let lock = NSLock()

    init(modelPath: String) throws {
        var cparams = whisper_context_default_params()
        cparams.use_gpu = true
        guard let c = whisper_init_from_file_with_params(modelPath, cparams) else {
            throw NSError(domain: "Whisp.Transcriber", code: 1,
                          userInfo: [NSLocalizedDescriptionKey: "Failed to load model at \(modelPath)"])
        }
        ctx = c
    }

    deinit { whisper_free(ctx) }

    func transcribe(_ samples: [Float]) async throws -> String {
        lock.lock(); defer { lock.unlock() }
        var params = whisper_full_default_params(WHISPER_SAMPLING_GREEDY)
        params.language = ("en" as NSString).utf8String
        params.translate = false
        params.print_progress = false
        params.print_realtime = false
        params.no_timestamps = true

        let rc = samples.withUnsafeBufferPointer { buf in
            whisper_full(ctx, params, buf.baseAddress, Int32(buf.count))
        }
        guard rc == 0 else {
            throw NSError(domain: "Whisp.Transcriber", code: 2,
                          userInfo: [NSLocalizedDescriptionKey: "whisper_full failed (\(rc))"])
        }
        var text = ""
        for i in 0..<whisper_full_n_segments(ctx) {
            if let seg = whisper_full_get_segment_text(ctx, i) {
                text += String(cString: seg)
            }
        }
        return text.trimmingCharacters(in: .whitespacesAndNewlines)
    }
}
```

```swift
import Foundation
import AVFoundation

enum WavReader {
    /// Reads any audio file and returns 16kHz mono Float samples.
    static func readMono16k(path: String) throws -> [Float] {
        let url = URL(fileURLWithPath: path)
        let file = try AVAudioFile(forReading: url)
        let outFormat = AVAudioFormat(commonFormat: .pcmFormatFloat32,
                                      sampleRate: 16000, channels: 1, interleaved: false)!
        let converter = AVAudioConverter(from: file.processingFormat, to: outFormat)!
        let inBuf = AVAudioPCMBuffer(pcmFormat: file.processingFormat,
                                     frameCapacity: AVAudioFrameCount(file.length))!
        try file.read(into: inBuf)

        let ratio = 16000.0 / file.processingFormat.sampleRate
        let outCap = AVAudioFrameCount(Double(inBuf.frameLength) * ratio) + 1024
        let outBuf = AVAudioPCMBuffer(pcmFormat: outFormat, frameCapacity: outCap)!
        var fed = false
        var err: NSError?
        converter.convert(to: outBuf, error: &err) { _, status in
            if fed { status.pointee = .noDataNow; return nil }
            fed = true; status.pointee = .haveData; return inBuf
        }
        if let err { throw err }
        let ptr = outBuf.floatChannelData![0]
        return Array(UnsafeBufferPointer(start: ptr, count: Int(outBuf.frameLength)))
    }
}
```

```swift
import Foundation
import AVFoundation

final class AudioRecorder {
    private let engine = AVAudioEngine()
    private var buffers: [AVAudioPCMBuffer] = []
    private var inputFormat: AVAudioFormat?
    var level: (Float) -> Void = { _ in }

    func start() throws {
        buffers.removeAll()
        let input = engine.inputNode
        let fmt = input.inputFormat(forBus: 0)
        inputFormat = fmt
        input.installTap(onBus: 0, bufferSize: 1024, format: fmt) { [weak self] buf, _ in
            guard let self else { return }
            if let copy = buf.copy() as? AVAudioPCMBuffer { self.buffers.append(copy) }
            self.level(Self.rms(buf))
        }
        engine.prepare()
        try engine.start()
    }

    /// Stops capture and returns 16kHz mono Float samples.
    func stop() -> [Float] {
        engine.inputNode.removeTap(onBus: 0)
        engine.stop()
        guard let inFmt = inputFormat, !buffers.isEmpty else { return [] }
        let outFormat = AVAudioFormat(commonFormat: .pcmFormatFloat32,
                                      sampleRate: 16000, channels: 1, interleaved: false)!
        guard let converter = AVAudioConverter(from: inFmt, to: outFormat) else { return [] }

        var out: [Float] = []
        for inBuf in buffers {
            let ratio = 16000.0 / inFmt.sampleRate
            let cap = AVAudioFrameCount(Double(inBuf.frameLength) * ratio) + 256
            guard let outBuf = AVAudioPCMBuffer(pcmFormat: outFormat, frameCapacity: cap) else { continue }
            var fed = false
            var err: NSError?
            converter.convert(to: outBuf, error: &err) { _, status in
                if fed { status.pointee = .noDataNow; return nil }
                fed = true; status.pointee = .haveData; return inBuf
            }
            if err == nil, let ptr = outBuf.floatChannelData?[0] {
                out.append(contentsOf: UnsafeBufferPointer(start: ptr, count: Int(outBuf.frameLength)))
            }
        }
        return out
    }

    private static func rms(_ buf: AVAudioPCMBuffer) -> Float {
        guard let ch = buf.floatChannelData?[0] else { return 0 }
        let n = Int(buf.frameLength)
        guard n > 0 else { return 0 }
        var sum: Float = 0
        for i in 0..<n { sum += ch[i] * ch[i] }
        return min(1, sqrt(sum / Float(n)) * 4)
    }
}
```

### Task 9: HotkeyMonitor (Fn hold-to-talk)

**Files:**
- Create: `Sources/Whisp/HotkeyMonitor.swift`

**Interfaces:**
- Consumes: nothing from Core.
- Produces: `final class HotkeyMonitor { init(onPress: @escaping () -> Void, onRelease: @escaping () -> Void); func start() }` — a CGEventTap that fires `onPress` when Fn goes down and `onRelease` when Fn goes up, with a 150ms debounce.

- [ ] **Step 1: Write `Sources/Whisp/HotkeyMonitor.swift`**

```swift
import AppKit
import CoreGraphics

final class HotkeyMonitor {
    private let onPress: () -> Void
    private let onRelease: () -> Void
    private var tap: CFMachPort?
    private var fnDown = false
    private var lastChange = Date.distantPast

    init(onPress: @escaping () -> Void, onRelease: @escaping () -> Void) {
        self.onPress = onPress
        self.onRelease = onRelease
    }

    func start() {
        let mask = (1 << CGEventType.flagsChanged.rawValue)
        let cb: CGEventTapCallBack = { _, _, event, refcon in
            let monitor = Unmanaged<HotkeyMonitor>.fromOpaque(refcon!).takeUnretainedValue()
            monitor.handle(event)
            return Unmanaged.passUnretained(event)
        }
        guard let tap = CGEvent.tapCreate(
            tap: .cgSessionEventTap, place: .headInsertEventTap,
            options: .listenOnly, eventsOfInterest: CGEventMask(mask),
            callback: cb, userInfo: Unmanaged.passUnretained(self).toOpaque()) else {
            NSLog("Whisp: failed to create event tap — grant Accessibility permission.")
            return
        }
        self.tap = tap
        let src = CFMachPortCreateRunLoopSource(kCFAllocatorDefault, tap, 0)
        CFRunLoopAddSource(CFRunLoopGetMain(), src, .commonModes)
        CGEvent.tapEnable(tap: tap, enable: true)
    }

    private func handle(_ event: CGEvent) {
        let flags = event.flags
        let isFn = flags.contains(.maskSecondaryFn)
        guard isFn != fnDown else { return }
        let now = Date()

        fnDown = isFn
        DispatchQueue.main.async { isFn ? self.onPress() : self.onRelease() }
    }
}
```

- [ ] **Step 1: Write `Sources/Whisp/PasteInserter.swift`**

```swift
import AppKit
import CoreGraphics
import WhispCore

final class PasteInserter: TextInserting {
    func insert(_ text: String) throws {
        guard !text.isEmpty else { return }
        let pb = NSPasteboard.general
        let snap = ClipboardPlan.snapshot(currentString: pb.string(forType: .string))

        pb.clearContents()
        pb.setString(text, forType: .string)

        try postCommandV()

        // Restore the previous clipboard shortly after the paste lands.
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.2) {
            pb.clearContents()
            if let prior = ClipboardPlan.restoreString(from: snap) {
                pb.setString(prior, forType: .string)
            }
        }
    }

    private func postCommandV() throws {
        guard let src = CGEventSource(stateID: .combinedSessionState) else {
            throw NSError(domain: "Whisp.Inserter", code: 1,
                          userInfo: [NSLocalizedDescriptionKey: "No event source"])
        }
        let vKey: CGKeyCode = 9 // 'v'
        let down = CGEvent(keyboardEventSource: src, virtualKey: vKey, keyDown: true)
        down?.flags = .maskCommand
        let up = CGEvent(keyboardEventSource: src, virtualKey: vKey, keyDown: false)
        up?.flags = .maskCommand
        down?.post(tap: .cgAnnotatedSessionEventTap)
        up?.post(tap: .cgAnnotatedSessionEventTap)
    }
}
```

**Interfaces:**

- [ ] **Step 1: Write `Sources/Whisp/DeepSeekCleaner.swift`**

```swift
import Foundation
import WhispCore

final class DeepSeekCleaner: Cleaning, @unchecked Sendable {
    private let apiKey: String
    private let endpoint = URL(string: "https://api.deepseek.com/chat/completions")!

    init(apiKey: String) { self.apiKey = apiKey }

    struct Req: Encodable { let model: String; let messages: [ChatMessage]; let stream: Bool }
    struct Resp: Decodable {
        struct Choice: Decodable { struct Msg: Decodable { let content: String }; let message: Msg }
        let choices: [Choice]
    }

    func complete(_ messages: [ChatMessage]) async throws -> String {
        var req = URLRequest(url: endpoint)
        req.httpMethod = "POST"
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")
        req.setValue("Bearer \(apiKey)", forHTTPHeaderField: "Authorization")
        req.httpBody = try JSONEncoder().encode(
            Req(model: "deepseek-chat", messages: messages, stream: false))
        req.timeoutInterval = 20

        let (data, resp) = try await URLSession.shared.data(for: req)
        guard let http = resp as? HTTPURLResponse, (200..<300).contains(http.statusCode) else {
            let code = (resp as? HTTPURLResponse)?.statusCode ?? -1
            throw NSError(domain: "Whisp.DeepSeek", code: code,
                          userInfo: [NSLocalizedDescriptionKey: "HTTP \(code)"])
        }
        let decoded = try JSONDecoder().decode(Resp.self, from: data)
        return decoded.choices.first?.message.content ?? ""
    }
}
```

```swift
            label.centerXAnchor.constraint(equalTo: card.centerXAnchor),
            meter.leadingAnchor.constraint(equalTo: card.leadingAnchor, constant: 16),
            meter.trailingAnchor.constraint(equalTo: card.trailingAnchor, constant: -16),
            meter.bottomAnchor.constraint(equalTo: card.bottomAnchor, constant: -10),
        ])
        positionBottomCenter()
    }

    private func positionBottomCenter() {
        guard let screen = NSScreen.main else { return }
        let f = screen.visibleFrame
        window.setFrameOrigin(NSPoint(x: f.midX - 120, y: f.minY + 80))
    }

    func setLevel(_ level: Float) {
        DispatchQueue.main.async { self.meter.doubleValue = Double(level) }
    }

    func show(_ status: PipelineStatus) {
        DispatchQueue.main.async {
            self.hideWork?.cancel()
            switch status {
            case .idle:        self.window.orderOut(nil); return
            case .listening:   self.label.stringValue = "🎙 Listening…"; self.meter.isHidden = false
            case .transcribing:self.label.stringValue = "✍️ Transcribing…"; self.meter.isHidden = true
            case .polishing:   self.label.stringValue = "✨ Polishing…"; self.meter.isHidden = true
            case .done:        self.label.stringValue = "✓ Done"; self.meter.isHidden = true; self.scheduleHide()
            case .error(let m):self.label.stringValue = "⚠️ \(m)"; self.meter.isHidden = true; self.scheduleHide()
            }
            self.positionBottomCenter()
            self.window.orderFrontRegardless()
        }
    }

    private func scheduleHide() {
        let w = DispatchWorkItem { [weak self] in self?.window.orderOut(nil) }
        hideWork = w
        DispatchQueue.main.asyncAfter(deadline: .now() + 1.2, execute: w)
    }
}
```

- **Edit Snippets…** / **Settings…** from the menu bar.

- `WhispCore` — pure logic (config, snippets, cleaner prompts, `SpeechPipeline`), unit-tested.
- `Whisp` — menu-bar app: AVAudioEngine recorder, vendored whisper.cpp transcriber,
